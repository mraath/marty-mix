function callout(text, type) {
	const allText = `> [!${type}]\n` + text;
	const lines = allText.split('\n');
	return lines.join('\n> ') + '\n';
}

// Get today's date and one month ago
const now = moment();
const lastMonth = moment().subtract(1, "month");

// Collect all undone tasks from files created OR updated in the last month
const tasks = dv.pages()
	.filter(p =>
		(p.created && moment(p.created).isAfter(lastMonth)) ||
		(p.updated && moment(p.updated).isAfter(lastMonth))
	)
	.flatMap(p => p.file.tasks)
	.filter(t => !t.completed);

// Display them in a nice callout
if (tasks.length === 0) {
	dv.paragraph(callout("No outstanding todos from the last month 🎉", "info"));
} else {
	let text = tasks.map(t => `- [ ] ${t.text} _(in [[${t.path}]])_`).join("\n");
	dv.paragraph(callout(text, "todo"));
}
